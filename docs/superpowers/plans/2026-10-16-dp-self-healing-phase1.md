```python
# deploy/failure_types.py
"""Failure kinds shared by the DP refresh summary and the Parts Availability alert (rendered into
Run_DP_Refresh and Notify_PartsAvailability ahead of the modules that use it).

failure_kind(error) reads a Fabric error message and returns one kind. The alerts turn the kind into a cause and
an exact fix; the later self-healing phases use it to decide what may be retried automatically.
Spec: data-projects/docs/superpowers/specs/2026-10-16-dp-self-healing-refresh-design.md
"""

# (kind, lower-case substrings). First match wins, so the more specific kinds come first: an SPN refusal says
# "not allowed", a credentials error mentions "Odbc", and a gateway timeout mentions the gateway.
FAILURE_PATTERNS = [
    ("unclaimed", ("spnbasedrefreshnotallowed", "spn based refresh is not allowed")),
    ("overlap", ("already in progress", "another refresh", "concurrent refresh", "currently being refreshed")),
    ("credentials", ("credential", "unauthorized", "not authorized", "forbidden", "access denied", "accessdenied",
                     "login failed", "password", "permission", "aadsts", "access token")),
    ("timeout", ("timeout", "timed out", "time out")),
    ("capacity", ("capacity", "throttl", "too many requests")),
    ("gateway", ("evaluationgatewayunreachable", "on-premises data gateway", "clouddatahub.net", "gateway",
                 "dm_gwpipeline", "odbc", "equiprdb64", "sql anywhere", "communication link")),
]
# Kinds a later phase may retry once without a person (spec 2c).
AUTO_RETRY_KINDS = {"timeout", "capacity"}


def failure_kind(error) -> str:
    """The first matching kind in FAILURE_PATTERNS, or "unknown"."""
    low = (error or "").lower()
    for kind, needles in FAILURE_PATTERNS:
        if any(n in low for n in needles):
            return kind
    return "unknown"
```

def fix_lines(items: dict, failed: list) -> list:
